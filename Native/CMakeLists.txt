cmake_minimum_required(VERSION 3.20)
project(SpatialFabricAudio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(SpatialFabricAudio SHARED src/audio.cpp)
target_include_directories(SpatialFabricAudio PUBLIC include PRIVATE src)
if(WIN32)
  target_link_libraries(SpatialFabricAudio PRIVATE ole32 uuid)
elseif(APPLE)
  target_link_libraries(SpatialFabricAudio PRIVATE "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(SpatialFabricAudio PRIVATE Threads::Threads)
endif()
add_executable(sf_audio_tests tests/audio_tests.cpp)
target_include_directories(sf_audio_tests PRIVATE include src)
target_link_libraries(sf_audio_tests PRIVATE SpatialFabricAudio)
enable_testing()
add_test(NAME silent_native_audio COMMAND sf_audio_tests)
